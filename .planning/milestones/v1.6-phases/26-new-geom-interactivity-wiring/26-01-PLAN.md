    ```javascript
          .attr('y2', flip ? xScaleFunc(d.xmax) : yScaleFunc(d.ymax));

          .attr('x1', flip ? yScaleFunc(d.ymin) : xScaleFunc(d.xmin))

          .attr('y1', flip ? xScaleFunc(d.xmin) : yScaleFunc(d.ymin))
          .attr('y2', flip ? xScaleFunc(d.xmax) : yScaleFunc(d.ymin));

    // geom_interval: linerange
    container.selectAll('g.geom-interval-linerange g.interval-item')

          .attr('cy', flip ? xScaleFunc(d.x) : yScaleFunc(d.y));
      });
    ```

    Key design decisions (from RESEARCH.md):
    - Use scoped selectors (`g.geom-interval-errorbar g.interval-item`) NOT bare `g.interval-item` to avoid cross-contamination between geom sub-types (Pitfall 4 — linerange/pointrange lack xmin/xmax fields).
    - Apply `.transition(t)` on each child element, NOT on the parent `g` (Pitfall 2 — D3 transitions on `<g>` don't propagate to child positional attrs).
    - Use `xScaleFunc`/`yScaleFunc` (already flip-swapped at line 206-207) and mirror the interval.js render patterns for coordinate mapping (Pitfall 3).
    - Use `d.x` and `d.y` as string keys since updateGeoms doesn't receive the layer aes object — consistent with how dotplot (line 333) and boxplot (line 302) updates work.

    IMPORTANT: The existing stub has `.transition(t)` on the selectAll BEFORE `.each()`. The replacement must NOT have `.transition(t)` before `.each()` — transitions go on each child selection inside the `.each()` callback.
