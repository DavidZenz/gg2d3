  /**
   * Attach tooltip event handlers to interactive elements.
   * Finds all geom elements matching INTERACTIVE_SELECTORS and adds
   * mouseover/mousemove/mouseout handlers that show/move/hide tooltip.
   */
  function attachTooltips(el, config, ir) {
    const svg = d3.select(el).select('svg');

    INTERACTIVE_SELECTORS.forEach(selector => {
      const selection = svg.selectAll(selector);

      // Skip if no elements match this selector
      if (selection.empty()) {
        return;
      }

      selection
        .on('mouseover.tooltip', function(event, d) {
          window.gg2d3.tooltip.show(event, d, config, ir);
        })
        .on('mousemove.tooltip', function(event) {
          window.gg2d3.tooltip.move(event);
        })
        .on('mouseout.tooltip', function() {
          window.gg2d3.tooltip.hide();
        });
    });
  }

**Hover pattern** (lines 558-640):
```javascript
  function attachHover(el, config) {
    const svg = d3.select(el).select('svg');

    INTERACTIVE_SELECTORS.forEach(selector => {
      const selection = svg.selectAll(selector);
      if (selection.empty()) {
        return;
      }

      selection
        .on('mouseover.hover', function(event, d) {
          var panelNode = this.closest('.panel');
          if (panelNode && panelNode.getAttribute('data-brush-active') === 'true') {
            return;
          }

          INTERACTIVE_SELECTORS.forEach(sel => {
            svg.selectAll(sel)
              .style('opacity', config.opacity);
          });

          const elem = d3.select(this);
          elem.style('opacity', 1.0);

          if (config.stroke) {
            elem.attr('data-hover-stroke', '1');
            elem.style('stroke', config.stroke);
            if (config.stroke_width) {
              elem.style('stroke-width', config.stroke_width);
            }
          }
        })
        .on('mouseout.hover', function() {
          /* restore opacity and stroke */
        });
    });
  }
```

The Phase 29 document should say sf hover reuses this behavior directly once `path.geom-sf` is in the selector list.

**Hit-test pattern to extend** (lines 255-319):
```javascript
  function isElementInPixelRect(node, rect) {
    var tagName = node.tagName.toLowerCase();

    if (tagName === 'circle') {
      var cx = parseFloat(node.getAttribute('cx'));
      var cy = parseFloat(node.getAttribute('cy'));
      return cx >= rect.px0 && cx <= rect.px1 &&
             cy >= rect.py0 && cy <= rect.py1;
    }

    if (tagName === 'path') {
      // Use bounding box center for path elements
      try {
        var bbox = node.getBBox();
        var centerX = bbox.x + bbox.width / 2;
        var centerY = bbox.y + bbox.height / 2;
        return centerX >= rect.px0 && centerX <= rect.px1 &&
               centerY >= rect.py0 && centerY <= rect.py1;
      } catch (e) {
        return false;
      }
    }

    return false;
  }
```

The future sf-specific design should prefer `data-cx`/`data-cy` when `node.matches('path.geom-sf')`, falling back to bounding-box center only for other paths.

**Selected data collection pattern** (lines 386-405):

  function collectSelectedData(panelGroup, pixelRect) {
    var clippedGroup = panelGroup.select('g[clip-path]');
